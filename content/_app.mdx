export const IIIF_SIZE_FALLBACK =
  // Canopy builds IIIF srcsets as `!w,w` at fixed widths (360–1600). The
  // Understory image server rejects `!w,h` whenever w or h exceeds the
  // image's own size, even though the result would be a downscale, so wide
  // or short images 400 on large screens. When one fails, read its info.json
  // once and clamp each candidate to the image's dimensions. Remove once
  // Canopy clamps srcsets upstream.
  `(function () {
  var tried = new WeakSet();
  function clamp(url, width, height) {
    var parts = url.split("/full/!");
    if (parts.length !== 2) return null;
    var rest = parts[1].split("/");
    var box = rest[0].split(",").map(Number);
    var w = Math.min(box[0], width);
    var h = Math.min(box[1], height);
    var scale = Math.min(w / width, h / height);
    rest[0] = w + "," + h;
    return {url: parts[0] + "/full/!" + rest.join("/"), width: Math.round(width * scale)};
  }
  document.addEventListener("error", function (event) {
    var img = event.target;
    if (!(img instanceof HTMLImageElement) || tried.has(img)) return;
    var failed = img.currentSrc || img.src;
    var at = failed.indexOf("/full/!");
    if (at < 0) return;
    tried.add(img);
    fetch(failed.slice(0, at) + "/info.json")
      .then(function (response) { return response.json(); })
      .then(function (info) {
        var src = clamp(img.getAttribute("src") || "", info.width, info.height);
        var seen = {};
        // IIIF sizes contain commas ("!360,360"), so split on ", " only.
        var candidates = (img.getAttribute("srcset") || "").split(", ").map(function (entry) {
          var fit = clamp(entry.trim().split(" ")[0], info.width, info.height);
          if (!fit || seen[fit.width]) return "";
          seen[fit.width] = true;
          return fit.url + " " + fit.width + "w";
        }).filter(Boolean);
        if (candidates.length) img.setAttribute("srcset", candidates.join(", "));
        if (src) img.setAttribute("src", src.url);
      })
      .catch(function () { img.removeAttribute("srcset"); });
  }, true);
})();`;

export const SEARCH_LABEL_TAGS =
  // Canopy's search script builds the results panel in the browser, so it
  // can't be dressed at build time. As it appears:
  // - tag each group label ("Works", "Pages") with its word; custom.css swaps
  //   in the drawn version (assets/images/glyphs/label-<word>.svg) for known
  //   words and leaves any other label in plain type;
  // - give each result a random hover annotation (1–30, from
  //   app/scripts/glyphs/annotations.py) and a random paint color. Rows in
  //   the header dropdown take a mark behind their text. On /search, whose
  //   results also render in the browser, each card takes a vertical mark
  //   down its left or right edge (never over its text): beside the image
  //   for works, beside the text card for pages.
  `(function () {
  var drawn = {works: 1, pages: 1, opere: 1, pagine: 1};
  var colors = ["red", "blue", "green", "yellow", "pink"];
  function tag() {
    var labels = document.querySelectorAll(".canopy-search-teaser__label:not([data-fx-label])");
    for (var i = 0; i < labels.length; i++) {
      var word = labels[i].textContent.trim().toLowerCase();
      labels[i].setAttribute("data-fx-label", drawn[word] ? word : "");
    }
    function mark(el) {
      el.setAttribute("data-fx-ann", 1 + Math.floor(Math.random() * 30));
      el.setAttribute("data-fx-ann-color", colors[Math.floor(Math.random() * colors.length)]);
    }
    var rows = document.querySelectorAll(".canopy-search-teaser__item:not([data-fx-ann])");
    for (var j = 0; j < rows.length; j++) mark(rows[j]);
    var cards = document.querySelectorAll(
      ".canopy-search-results .canopy-card-media:not([data-fx-ann]), " +
      ".canopy-search-results .canopy-card figure:not(:has(.canopy-card-media)):not([data-fx-ann]), " +
      ".canopy-search-results .canopy-article-card:not([data-fx-ann])"
    );
    for (var k = 0; k < cards.length; k++) {
      mark(cards[k]);
      cards[k].setAttribute("data-fx-ann-dir", "v");
      cards[k].setAttribute("data-fx-ann-side", Math.random() < 0.5 ? "left" : "right");
    }
  }
  new MutationObserver(tag).observe(document.documentElement, {childList: true, subtree: true});
})();`;

export const HEADER_TAGLINE = "Giuseppe Chiari at 100";

export const SECONDARY_SECTIONS =
  // Sections that sit in the quiet secondary menu beside search instead of
  // the main nav, matched on the last path segment (`/it/about`, `/search`).
  ["about", "search"];

export function navSection(href) {
  const path = String(href).split(/[?#]/)[0].replace(/\/(index\.html)?$/, "").replace(/\.html$/, "");
  return path.split("/").pop();
}

export const GENERATED_SECTIONS =
  // Sections Canopy builds for every locale from IIIF data or templates, so
  // their counterpart pages always exist without an MDX file.
  ["works", "search"];

export function counterpartHref(pathname, locale, locales) {
  // Same page in another locale: the Italian content mirrors the English
  // folder structure, so swap the /it prefix. Fall back to that locale's
  // home when the counterpart page doesn't exist.
  const prefixes = locales.filter((l) => !l.default).map((l) => l.lang);
  const parts = pathname.replace(/^\/+/, "").split("/");
  const rest = prefixes.includes(parts[0]) ? parts.slice(1) : parts;
  const target = [locale.default ? null : locale.lang, ...rest].filter(Boolean).join("/");
  const home = locale.default ? "/" : `/${locale.lang}`;
  if (!rest.join("")) return home;
  const generated = GENERATED_SECTIONS.includes(rest[0]);
  const mdx = `content/${target}`.replace(/\.html$/, ".mdx");
  const exists = generated || fs.existsSync(mdx) || fs.existsSync(mdx.replace(/\/index\.mdx$/, ".mdx"));
  return exists ? `/${target}` : home;
}

export function LanguageToggle({page, site}) {
  // English / Italiano drawn in Chiari's hand. The current language is full
  // black; the other is Rich Black 50% and links to the same page there.
  const {Hand} = useMDXComponents();
  const locales = (site && site.languageToggle && site.languageToggle.locales) || [];
  if (locales.length < 2) return null;
  const pathname = String(page.href || "/").split(/[?#]/)[0];
  const first = pathname.replace(/^\/+/, "").split("/")[0];
  const active = locales.find((l) => !l.default && l.lang === first) || locales.find((l) => l.default) || locales[0];
  return (
    <nav className="fx-lang" aria-label="Language">
      {locales.map((locale) => {
        const current = locale.lang === active.lang;
        const label = locale.label || locale.lang;
        return (
          <a
            key={locale.lang}
            href={withBasePath(counterpartHref(pathname, locale, locales))}
            hrefLang={locale.lang}
            lang={locale.lang}
            aria-current={current ? "true" : undefined}
          >
            {Hand ? <Hand as="span" size="0.95rem" tracking={8}>{label}</Hand> : label}
          </a>
        );
      })}
    </nav>
  );
}

export const MENU_PAINT =
  // Paint color for each main section, in navigation.yml order.
  ["red", "blue", "green", "yellow"];

export function paintShape(color) {
  // The stroke's filled outline, read from its SVG so the file stays the
  // single source for each color.
  const svg = fs.readFileSync(`assets/images/marker/paint-${color}.svg`, "utf8");
  return {fill: svg.match(/<g fill="([^"]+)"/)[1], paths: svg.match(/<g[^>]*>([\s\S]*)<\/g>/)[1]};
}

export function PaintStroke({color, id}) {
  // A menu stroke that is written rather than revealed: a round-capped line
  // along the stroke's centreline (the brush tip) masks the paint shape, so
  // paint appears only where the tip has passed. CSS animates the tip with
  // stroke-dashoffset; the bristle fibres at the tail arrive last.
  const {fill, paths} = paintShape(color);
  return (
    <svg className="fx-paint" viewBox="0 0 200 24" preserveAspectRatio="none" aria-hidden="true" focusable="false">
      <defs>
        <mask id={id} maskUnits="userSpaceOnUse" x="-12" y="-2" width="224" height="28">
          <path
            className="fx-paint__tip"
            d="M3 12.5C50 10 110 11 160 10.5S190 11 198 11"
            fill="none"
            stroke="#fff"
            strokeWidth="22"
            strokeLinecap="round"
            pathLength="1"
          />
        </mask>
      </defs>
      <g fill={fill} mask={`url(#${id})`} dangerouslySetInnerHTML={{__html: paths}} />
    </svg>
  );
}

export function ChiariHeader() {
  // Canopy's header with navigation.yml split in two: Watch, Play, Write and
  // Collaborate stay in the main nav row; About and Works move to a plain
  // secondary menu next to search. Every link stays in navigation.yml, so
  // Canopy still resolves locale labels, /it/ routes and the current page.
  // The page context is shared on globalThis (see getSafePageContext in
  // @canopy-iiif/app/ui).
  const PageContext = globalThis[Symbol.for("__CANOPY_PAGE_CONTEXT__")];
  const context = PageContext ? React.useContext(PageContext) : null;
  // Canopy leaves isActive unset on primary links, so mark the section the
  // current page lives under (e.g. /it/watch/per_arco/ → Watch).
  const page = (context && context.page) || {};
  const pageSegments = String(page.href || page.url || "").split(/[?#]/)[0].split("/");
  const links = ((context && context.primaryNavigation) || []).map((link) => ({
    ...link,
    isActive: link.isActive || pageSegments.includes(navSection(link.href)),
  }));
  const isSecondary = (link) => SECONDARY_SECTIONS.includes(navSection(link.href));
  const secondary = links.filter(isSecondary);
  // Each main label carries its own paint stroke; Canopy renders the label
  // as given inside its link.
  const main = links.filter((link) => !isSecondary(link)).map((link, i) => ({
    ...link,
    label: (
      <>
        {link.label || link.href}
        <PaintStroke color={MENU_PAINT[i % MENU_PAINT.length]} id={`fx-paint-${navSection(link.href)}`} />
      </>
    ),
  }));
  return (
    <div className="fx-masthead">
      {/* The header brand shows ART IS EASY (an image, via CSS) over this
          line of text; the page <title> keeps canopy.yml's "Chiari". */}
      <CanopyHeader
        title={HEADER_TAGLINE}
        navigation={main}
        searchPlaceholder=""
        languageToggle={false}
      />
      <div className="fx-subbar">
        {secondary.length ? (
          <nav className="fx-secondary-nav" aria-label="Secondary">
            {secondary.map((link) => (
              <a key={link.href} href={link.href} aria-current={link.isActive ? "page" : undefined}>
                {link.label || link.href}
              </a>
            ))}
          </nav>
        ) : null}
        <LanguageToggle page={page} site={(context && context.site) || {}} />
      </div>
    </div>
  );
}
